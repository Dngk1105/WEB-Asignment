## Cấu trúc thư mục Assignment 3
```
assignment-03/
├── index.html                  # Trang điều hướng tổng quan Assignment 3
├── README.md                   # Tài liệu báo cáo Assignment 3
│
├── bai-tap-1/                  # Bài tập 1: Tự sửa lỗi (không dùng AI)
│   ├── index.html              # Trang diff so sánh & phân tích lỗi CSS
│   ├── trang.html              # Trang web chạy với CSS đã sửa
│   ├── style_new.css           # CSS đã sửa hoàn thiện
│   ├── style_old.css           # Bản CSS gốc bị lỗi để đối chứng
│   ├── images/                 # Ảnh sản phẩm và hero
│   └── phan-tich-loi.md        # Báo cáo chi tiết phân tích lỗi
│
├── bai-tap-2/                  # Bài tập 2: Sửa lỗi dùng Prompt GenAI
│   ├── index.html              # Trang web chạy với CSS đã sửa
│   ├── trang.html              # File HTML gốc
│   ├── style.css               # CSS đã sửa theo kết quả GenAI
│   ├── style-loi-goc.css       # Bản CSS gốc bị lỗi để đối chứng
│   ├── images/                 # Ảnh sản phẩm và hero
│   └── prompt-ai.md            # Bộ Prompt chuyên gia & báo cáo giải trình
│
├── bai-tap-3/                  # Bài tập 3: Responsive Web Design Demo
│   ├── index.html              # Trang demo điều hướng
│   └── responsive-Bai tap 3.html # File demo bố cục 3 cột / 2 cột / 1 cột
│
└── bai-tap-4/                  # Bài tập 4: Xây dựng CSS Responsive (IDE Dark Theme)
    ├── index.html              # Trang hiển thị với CSS theme IDE
    ├── trang-goc.html          # File HTML gốc (giữ nguyên không sửa đổi)
    ├── style.css               # CSS hoàn chỉnh theo phong cách Dark IDE
    └── images/                 # Ảnh đội ngũ và tài nguyên minh hoạ
```
